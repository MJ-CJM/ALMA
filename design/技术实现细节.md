```python
def validate_invite_code(self, code: str) -> bool:
    """验证邀请码有效性 - 支持使用次数和过期时间"""
    try:
        with self.get_session() as session:
            invite_code = session.exec(
                select(InviteCode).where(InviteCode.code == code)
            ).first()
            
            if not invite_code:
                return False
            
            # 检查状态
            if invite_code.status != 'active':
                return False
            
            # 检查使用次数
            if invite_code.used_count >= invite_code.max_uses:
                return False
            
            # 检查过期时间
            if invite_code.expires_at and datetime.now() > invite_code.expires_at:
                return False
            
            return True
            
    except Exception as e:
        print(f"验证邀请码失败: {e}")
        return False
```

**安全特性**：

#### API 格式自动处理

```python
def run_dify_workflow_blocking(self, question: str, user_id: str, workflow_id: str) -> Dict:
    """调用 Dify 工作流 - 自动处理 API 格式"""
    workflow_config = self.get_workflow_by_id(workflow_id)
    if not workflow_config:
        raise ValueError(f"工作流 {workflow_id} 不存在")
    
    # 处理 API Key 格式：确保有 Bearer 前缀
    api_key = workflow_config['api_key'].strip()
    if not api_key.startswith('Bearer '):
        api_key = f"Bearer {api_key}" if api_key else api_key
    
    # 清理 api_base，确保没有尾部斜杠
    api_base = workflow_config['api_base'].rstrip('/')
    
    url = f"{api_base}/v1/workflows/run"
    headers = {
        "Authorization": api_key,
        "Content-Type": "application/json",
    }
    
    payload = {
        "inputs": {"question": question},
        "response_mode": "blocking",
        "user": user_id,
    }
    
    try:
        resp = requests.post(url, headers=headers, json=payload, timeout=120)
        resp.raise_for_status()
        return resp.json()
    except requests.RequestException as e:
        raise ConnectionError(f"Dify API 调用失败: {str(e)}")
```

**技术特点**：
- **数据库驱动**：工作流配置存储在数据库中，支持动态管理
- **权限控制**：支持全局可见和指定用户可见两种模式
- **向后兼容**：如果数据库为空，自动从 `config.json` 导入
- **API 格式处理**：自动处理 API Key 的 Bearer 前缀和 API Base 的尾部斜杠

```python
def parse_dify_response(self, data: Dict) -> str:
    """智能解析 Dify 响应 - 兼容多种响应格式"""
    try:
        # 尝试不同的响应结构
        payload = data.get("data") or data
        
        # 尝试从 outputs 中获取答案
        outputs = payload.get("outputs") or {}
        answer = (
            outputs.get("answer") or
            outputs.get("text") or
            outputs.get("result") or
            outputs.get("response")
        )
        
        if answer:
            return str(answer)
        
        # 尝试从 message 字段获取
        message = payload.get("message")
        if isinstance(message, str):
            return message
        
        # 兜底：返回整个 outputs 的 JSON
        if outputs:
            return json.dumps(outputs, ensure_ascii=False, indent=2)
        
        return "暂时无法生成回复，请稍后重试"
        
    except Exception as e:
        return f"解析回复失败: {str(e)}"
```

```python
def load_user_conversations():
    """加载用户对话历史 - 仅加载元数据，延迟加载消息"""
    # 检查是否已加载，以及是否是为当前用户加载的
    loaded_user_id = st.session_state.get("conversations_loaded_user_id")
    current_user_id = st.session_state.get("user_id")
    
    if st.session_state.get("conversations_loaded", False) and loaded_user_id == current_user_id:
        return
    
    try:
        user_id = st.session_state.user_id
        conversations = db_manager.load_user_conversations(user_id)
        
        # 仅存储对话元数据，不加载消息历史
        st.session_state.conversations = {}
        for conv in conversations:
            st.session_state.conversations[conv.conversation_name] = {
                "id": conv.id,
                "workflow_id": conv.workflow_id,
                "history": None,  # 标记为未加载
                "messages_loaded": False
            }
        
        st.session_state.conversations_loaded = True
        st.session_state.conversations_loaded_user_id = user_id
        
    except Exception as e:
        print(f"加载对话失败: {e}")

def load_conversation_messages(conv_name: str):
    """按需加载对话消息"""
    if conv_name not in st.session_state.conversations:
        return
    
    conv_info = st.session_state.conversations[conv_name]
    
    # 如果消息已经加载，直接返回
    if conv_info.get("messages_loaded", False):
        return
    
    # 加载消息
    try:
        messages = db_manager.load_conversation_messages(conv_info["id"])
        message_list = [
            {"role": msg.role, "content": msg.content}
            for msg in messages
        ]
        
        conv_info["history"] = message_list
        conv_info["messages_loaded"] = True
    except Exception as e:
        st.error(f"加载对话消息失败: {e}")
        conv_info["history"] = []
```

**性能优化策略**：
- **延迟加载**：登录时只加载对话列表，消息按需加载
- **实例缓存**：使用 `@st.cache_resource` 缓存 `DatabaseManager` 和 `AuthManager` 实例
- **唯一名称生成**：自动为同名对话添加后缀，避免覆盖
